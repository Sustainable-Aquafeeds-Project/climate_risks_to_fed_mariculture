Asian seabass

See the full calibration in 03_05_check_species_asian_seabass.

this_species <- "asian_seabass"
this_filename <- file.path(prepdata_path, "species_parameters", paste0("species_parameters_", this_species, ".qs"))

params <- read_excel(
  file.path(rawdata_path, 'parameters_species.xlsx'),
  sheet = this_species,
  trim_ws = T, n_max = 28
) %>%
  clean_names() %>% 
  select(used_in_script, value)

params_raw <- params$value
names(params_raw) <- params$used_in_script

params_calib <- if (file.exists(this_filename)) {qd_read(this_filename)} else {default}
